cmake_minimum_required(VERSION 3.20)
project(ptcrypt LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
find_package(cryptopp CONFIG QUIET)

add_executable(ptcrypt src/pt_lab_builder/ptcrypt.cpp)
if(TARGET cryptopp::cryptopp)
  target_link_libraries(ptcrypt PRIVATE cryptopp::cryptopp ZLIB::ZLIB)
elseif(TARGET cryptopp-static)
  target_link_libraries(ptcrypt PRIVATE cryptopp-static ZLIB::ZLIB)
else()
  find_path(CRYPTOPP_INCLUDE_DIR cryptopp/eax.h REQUIRED)
  find_library(CRYPTOPP_LIBRARY NAMES cryptopp cryptlib REQUIRED)
  target_include_directories(ptcrypt PRIVATE ${CRYPTOPP_INCLUDE_DIR})
  target_link_libraries(ptcrypt PRIVATE ${CRYPTOPP_LIBRARY} ZLIB::ZLIB)
endif()
